```typescript
function _compositorDecode(dataUrl: string): HTMLImageElement | null {
  const cached = _compositorImageCache.get(dataUrl);
  if (cached) return cached;
  if (_compositorImageLoading.has(dataUrl) || _compositorImageFailed.has(dataUrl)) return null;
  const image = new Image();

  };

```

**LRU entry shape** (from RESEARCH.md Pattern 2 — Claude's discretion on field names):
```typescript
interface DecodedFrameEntry {
  readonly key: string;        // stable frame ID: cachePath / sourceRef / library asset ID
  bitmap: ImageBitmap;
  readonly byteSize: number;   // width * height * 4 (decoded RGBA)
  pinCount: number;            // >0 = pinned (in-flight draw), unevictable
  lastUsed: number;            // monotonic LRU clock
}
```

**Note:** No dedicated LRU class exists in the codebase — this is a new utility. The Map-based cache idiom above is the closest analog. The LRU owns `bitmap.close()` on eviction (D-11); pinned entries (`pinCount > 0`) are skipped. Key by stable `sourceRef`/library asset ID/`cachePath`, NEVER by dataUrl (D-06). Budget in bytes (512 MB, D-09), never frame counts.

**PNG signature probe (D-07 retarget → WebP RIFF/VP8L)** (rotoCanvasFrames.ts:14-33):
```typescript
const ROTO_PNG_DATA_URL_HEADER = 'data:image/png;base64';
const PNG_SIGNATURE = [0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a] as const;
const ROTO_PNG_SIGNATURE_PROBE_CHARS = 40;

export function isRotoPngDataUrl(value: unknown): value is string {
  if (typeof value !== 'string') return false;
  const commaIndex = value.indexOf(',');
  if (commaIndex < 0 || value.slice(0, commaIndex).toLowerCase() !== ROTO_PNG_DATA_URL_HEADER) return false;
  try {
    const decoded = atob(value.slice(commaIndex + 1, commaIndex + 1 + ROTO_PNG_SIGNATURE_PROBE_CHARS).replace(/\s/g, ''));
    return decoded.length >= PNG_SIGNATURE.length
      && PNG_SIGNATURE.every((byte, index) => decoded.charCodeAt(index) === byte);
  } catch {
    return false;
  }
}
```

```typescript
export function getPhysicsPaintWorkingSize(projectWidth: number, projectHeight: number): { width: number; height: number } {
  if (projectWidth <= 0 || projectHeight <= 0) return { width: DEFAULT_PHYSICS_PAINT_CANVAS_WIDTH, height: DEFAULT_PHYSICS_PAINT_CANVAS_HEIGHT };

    height: Math.max(1, Math.round(projectHeight * scale)),
  };
}
```
